### 改进4: 技能缓存增强
```python
# 文件: MimirAether/agent/prompt_builder.py
# 添加磁盘快照:

_SKILLS_SNAPSHOT_VERSION = 1
_SKILLS_SNAPSHOT_PATH = None

def _get_skills_snapshot_path():
    global _SKILLS_SNAPSHOT_PATH
    if _SKILLS_SNAPSHOT_PATH is None:
        _SKILLS_SNAPSHOT_PATH = Path.home() / ".openclaw" / ".skills_snapshot.json"
    return _SKILLS_SNAPSHOT_PATH

def _load_skills_snapshot(skills_dir: Path) -> Optional[dict]:
    snapshot_path = _get_skills_snapshot_path()
    if not snapshot_path.exists():
        return None
    try:
        snapshot = json.loads(snapshot_path.read_text())
        if snapshot.get("version") != _SKILLS_SNAPSHOT_VERSION:
            return None
        manifest = _build_skills_manifest(skills_dir)
        if snapshot.get("manifest") != manifest:
            return None
        return snapshot
    except Exception:
        return None

def _write_skills_snapshot(skills_dir: Path, entries: list, categories: dict):
    manifest = _build_skills_manifest(skills_dir)
    snapshot = {
        "version": _SKILLS_SNAPSHOT_VERSION,
        "manifest": manifest,
        "skills": entries,
        "category_descriptions": categories,
    }
    try:
        _get_skills_snapshot_path().write_text(json.dumps(snapshot))
    except Exception as e:
        logger.debug(f"Failed to write skills snapshot: {e}")
```
